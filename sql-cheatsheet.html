SELECT & Filtering

The foundation of every SQL query — selecting, filtering, and shaping data.

WHEREIS NULLCASE WHEN NULL Handling & Conditional Filtering
⚡ NULL gotcha: NULL != 2 evaluates to NULL, not TRUE. Always explicitly use IS NULL or IS NOT NULL — never compare with = NULL.
Syntax Template
sql
SELECT col1, col2
FROM tbl
WHERE condition1
  AND (condition2 OR condition3)
  AND col IS NOT NULL;   -- always use IS NULL, not = NULL

-- NULL-safe tricks
col IS NULL              -- matches NULLs
col IS NOT NULL          -- excludes NULLs
COALESCE(col, default)   -- replace NULL with a fallback value
IFNULL(col, default)     -- MySQL alias for COALESCE with 2 args

Return NULL if there is no second highest — not an empty result set.

    FROM Employee

Aggregations & GROUP BY

Summarizing and grouping data — the most common LeetCode SQL pattern.

COUNTSUMAVGGROUP BYHAVING Aggregate Functions & GROUP BY
⚠️ HAVING vs WHERE: WHERE filters rows before grouping. HAVING filters groups after aggregation. You cannot use aggregate functions inside WHERE.
Syntax Template
sql
SELECT
    group_col,
    COUNT(*)               AS total_rows,
    COUNT(col)             AS non_null_count,   -- ignores NULLs
    COUNT(DISTINCT col)    AS unique_count,
    SUM(amount)            AS total,
    AVG(amount)            AS average,
    MIN(amount)            AS minimum,
    MAX(amount)            AS maximum,
    -- Conditional count shortcut (TRUE=1, FALSE=0 in MySQL):
    SUM(col = 'value')     AS matching_count,
    SUM(CASE WHEN col = 'value' THEN amount ELSE 0 END) AS cond_sum
FROM tbl
WHERE pre_filter_condition         -- filters BEFORE grouping
GROUP BY group_col
HAVING aggregate_condition         -- filters AFTER grouping
ORDER BY total_rows DESC;

2 Monthly Transactions Summary Medium LC #1193

Multi-column aggregation with a conditional sum — extremely common in LC database problems.

mysql · LC 1193
-- Transactions(id, country, state, amount, trans_date)
-- state ∈ {'approved', 'declined'}

SELECT
    DATE_FORMAT(trans_date, '%Y-%m')          AS month,
    country,
    COUNT(*)                                   AS trans_count,
    SUM(state = 'approved')                    AS approved_count,    -- boolean trick
    SUM(amount)                                AS trans_total_amount,
    SUM(CASE WHEN state = 'approved'
             THEN amount ELSE 0 END)           AS approved_total_amount
FROM Transactions
GROUP BY month, country;

JOINs

The #1 most tested SQL topic in interviews. Master every join type.

Join Type Quick Reference
INNER JOIN
Only rows with a match in both tables
LEFT JOIN
All left rows + matching right (NULL if no match)
RIGHT JOIN
All right rows + matching left (NULL if no match)
CROSS JOIN
Cartesian product — every row × every row
SELF JOIN
Table joined to itself via two aliases
LEFT JOININNER JOINAnti-Join JOIN Fundamentals & Anti-Join Pattern
⚡ Anti-Join: LEFT JOIN … WHERE right.id IS NULL finds rows that exist in the left table but have no matching row in the right table. Faster than NOT IN with NULLs.
Syntax Template
sql
-- INNER JOIN: matching rows only
SELECT a.col, b.col
FROM A a INNER JOIN B b ON a.id = b.a_id;

-- LEFT JOIN: all rows from A, NULL on B side if no match
SELECT a.col, b.col
FROM A a LEFT JOIN B b ON a.id = b.a_id;

-- Anti-Join: rows in A with NO match in B
SELECT a.col
FROM A a
LEFT JOIN B b ON a.id = b.a_id
WHERE b.a_id IS NULL;         -- ← the anti-join filter

Subqueries & CTEs

Modular, readable query building. CTEs are your best tool for complex multi-step problems.

INEXISTSNOT EXISTSCorrelated Subqueries: IN vs EXISTS
⚠️ NOT IN + NULL trap: If the subquery result contains any NULL, NOT IN returns zero rows. Use NOT EXISTS or the anti-join pattern instead for safety.
Syntax Template
sql
-- Scalar subquery (returns single value)
WHERE salary = (SELECT MAX(salary) FROM Employee)

-- IN subquery (returns list of values)
WHERE dept_id IN (SELECT id FROM Dept WHERE budget > 50000)

-- EXISTS (TRUE if subquery has any rows — often faster)
WHERE EXISTS (
    SELECT 1 FROM Orders WHERE Orders.customer_id = Customer.id
)

-- NOT EXISTS (safer than NOT IN when NULLs possible)
WHERE NOT EXISTS (
    SELECT 1 FROM Orders WHERE Orders.customer_id = Customer.id
)

-- Correlated subquery (references outer query row by row)
WHERE salary > (
    SELECT AVG(salary)
    FROM Employee e2
    WHERE e2.dept_id = e1.dept_id   -- ← references outer alias e1
)

-- Tuple / row comparison (MySQL extension)
WHERE (dept_id, salary) IN (SELECT dept_id, MAX(salary) FROM Employee GROUP BY dept_id)
1 Department Highest Salary Medium LC #184

The tuple (dept_id, salary) IN (...) trick elegantly filters for the max-salary employees per department.

mysql · LC 184
-- Employee(id, name, salary, departmentId)
-- Department(id, name)

SELECT d.name AS Department, e.name AS Employee, e.salary AS Salary
FROM Employee e
JOIN Department d ON e.departmentId = d.id
WHERE (e.departmentId, e.salary) IN (
    SELECT departmentId, MAX(salary)
    FROM Employee
    GROUP BY departmentId
);
2 Trips & Users — Multi-Step CTE Hard LC #262

CTEs let you name and reuse intermediate result sets — essential for breaking down hard problems into clear steps.

mysql · LC 262 — CTE approach
-- Trips(id, client_id, driver_id, status, request_at)
-- Users(users_id, banned, role)

WITH non_banned AS (
    -- Step 1: collect non-banned user IDs
    SELECT users_id FROM Users WHERE banned = 'No'
),
valid_trips AS (
    -- Step 2: keep only trips involving non-banned users in date range
    SELECT *
    FROM Trips
    WHERE request_at BETWEEN '2013-10-01' AND '2013-10-03'
      AND client_id IN (SELECT users_id FROM non_banned)
      AND driver_id IN (SELECT users_id FROM non_banned)
)
-- Step 3: compute daily cancellation rate
SELECT
    request_at                                   AS Day,
    ROUND(SUM(status != 'completed') / COUNT(*), 2) AS 'Cancellation Rate'
FROM valid_trips
GROUP BY request_at;

Window Functions

Compute across rows without collapsing. Essential for Medium/Hard ranking & analytics problems.

Window Function Skeleton
sql · skeleton
function_name() OVER (
    PARTITION BY col           -- defines "window" groups (optional)
    ORDER BY col DESC          -- ordering within each partition (optional)
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW  -- frame (optional)
)

-- ── Ranking functions ──────────────────────────────────────────────
ROW_NUMBER()       -- unique sequential number: 1,2,3,4 (no ties)
RANK()             -- ties share rank, next rank skips: 1,2,2,4
DENSE_RANK()       -- ties share rank, no gaps:         1,2,2,3  ← Nth highest

-- ── Navigation functions ───────────────────────────────────────────
LAG(col, n, default)    -- value n rows BEFORE current row
LEAD(col, n, default)   -- value n rows AFTER current row
FIRST_VALUE(col)        -- first value in the window partition
LAST_VALUE(col)         -- last value in the window partition

-- ── Aggregate as windows (rows not collapsed!) ─────────────────────
SUM(col)   OVER (...)   -- running total
AVG(col)   OVER (...)   -- rolling average
COUNT(col) OVER (...)   -- running count
NTILE(n)   OVER (...)   -- divide rows into n equal buckets (1..n)
DENSE_RANKROW_NUMBERPARTITION BY Ranking Functions
⚡ Use DENSE_RANK() for "Nth highest" problems (handles ties). Use ROW_NUMBER() to deduplicate or paginate.
1 Rank Scores (No Gaps) Medium LC #178

Tie scores must share the same rank with no gaps — exactly what DENSE_RANK() does.

mysql · LC 178
-- Scores(id, score)

SELECT
    score,
    DENSE_RANK() OVER (ORDER BY score DESC) AS 'rank'
FROM Scores
ORDER BY score DESC;

-- Comparison:
-- score:       100, 90, 90, 80
-- RANK():        1,  2,  2,  4   (gap after tie)
-- DENSE_RANK():  1,  2,  2,  3   (no gap — LC wants this)
-- ROW_NUMBER():  1,  2,  3,  4   (always unique)
2 Department Top 3 Salaries Hard LC #185

PARTITION BY departmentId resets the rank counter for each department independently.

mysql · LC 185
-- Employee(id, name, salary, departmentId)
-- Department(id, name)

WITH ranked AS (
    SELECT
        name, salary, departmentId,
        DENSE_RANK() OVER (
            PARTITION BY departmentId
            ORDER BY salary DESC
        ) AS rnk
    FROM Employee
)
SELECT d.name AS Department, r.name AS Employee, r.salary AS Salary
FROM ranked r
JOIN Department d ON r.departmentId = d.id
WHERE r.rnk <= 3;

1 Day-over-Day Page View Change Medium

LAG(col, 1, 0) returns the previous row's value. The third argument is the default when there is no previous row.

mysql · LAG pattern
-- Activity(user_id, activity_date, page_views)

SELECT
    user_id,
    activity_date,
    page_views,
    LAG(page_views, 1, 0) OVER (
        PARTITION BY user_id
        ORDER BY activity_date
    )                        AS prev_day_views,
    page_views - LAG(page_views, 1, 0) OVER (
        PARTITION BY user_id
        ORDER BY activity_date
    )                        AS daily_change
FROM Activity;

-- LAG(col, offset, default_when_no_prior_row)
-- LEAD(col, offset, default_when_no_next_row)  ← same but looks forward
2 Cumulative Revenue & 7-Day Rolling Average Medium
mysql · SUM OVER pattern
-- Sales(sale_id, sale_date, amount)

SELECT
    sale_id,
    sale_date,
    amount,
    -- Running total from beginning up to current row
    SUM(amount) OVER (
        ORDER BY sale_date
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    )                        AS running_total,
    -- Rolling 7-day average (current + 6 prior rows)
    ROUND(AVG(amount) OVER (
        ORDER BY sale_date
        ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
    ), 2)                    AS rolling_7day_avg
FROM Sales
ORDER BY sale_date;

SET Operations

Combine, intersect, or subtract result sets from multiple queries.

UNIONUNION ALLINTERSECTEXCEPT UNION, INTERSECT & EXCEPT
⚠️ MySQL version note: INTERSECT and EXCEPT are available in MySQL 8.0+. For older versions simulate with INNER JOIN and LEFT JOIN … IS NULL.
Syntax Template
sql
-- UNION: combine results, remove duplicates
SELECT col FROM A
UNION
SELECT col FROM B;

-- UNION ALL: combine results, KEEP duplicates (faster, use when you know no dups)
SELECT col FROM A
UNION ALL
SELECT col FROM B;

-- INTERSECT: rows in BOTH result sets (MySQL 8.0+)
SELECT col FROM A
INTERSECT
SELECT col FROM B;

-- EXCEPT: rows in first set but NOT in second (MySQL 8.0+)
SELECT col FROM A
EXCEPT
SELECT col FROM B;

-- Rules: same number of columns, compatible data types in each position

2 Non-Friends (EXCEPT simulation) Medium

Finding "everyone except a specific set" — show both the MySQL 8.0+ EXCEPT syntax and the backward-compatible anti-join.

mysql · EXCEPT & fallback
-- Users(id, name)  Friends(user1_id, user2_id)
-- People who are NOT friends with user_id = 1

-- MySQL 8.0+:
SELECT id, name FROM Users WHERE id != 1
EXCEPT
SELECT u.id, u.name
FROM Users u
JOIN Friends f
  ON (f.user1_id = 1 AND f.user2_id = u.id)
  OR (f.user2_id = 1 AND f.user1_id = u.id);

-- Backward-compatible anti-join (works on all MySQL versions):
SELECT u.id, u.name
FROM Users u
LEFT JOIN Friends f
  ON (f.user1_id = 1 AND f.user2_id = u.id)
  OR (f.user2_id = 1 AND f.user1_id = u.id)
WHERE u.id != 1
  AND f.user1_id IS NULL;   -- no friendship row found

Advanced Patterns

Hard-level tricks that separate good candidates from great ones.

Gaps & IslandsConsecutiveStreak Consecutive Rows & Streak Detection
⚡ Classic trick: Subtract ROW_NUMBER() from a date sequence — rows in the same consecutive group produce the same difference (the "island" key).
1 Consecutive Numbers (at least 3 in a row) Medium LC #180
mysql · LC 180
-- Logs(id, num)   id is sequential

-- Method 1: Triple self-join (simple, works for exactly 3)
SELECT DISTINCT l1.num AS ConsecutiveNums
FROM Logs l1
JOIN Logs l2 ON l2.id = l1.id + 1 AND l2.num = l1.num
JOIN Logs l3 ON l3.id = l1.id + 2 AND l3.num = l1.num;

-- Method 2: LAG/LEAD (scalable to N consecutive)
WITH flagged AS (
    SELECT num,
           LAG(num, 1) OVER (ORDER BY id) AS p1,
           LAG(num, 2) OVER (ORDER BY id) AS p2
    FROM Logs
)
SELECT DISTINCT num AS ConsecutiveNums
FROM flagged
WHERE num = p1 AND num = p2;
2 Longest Consecutive Login Streak Hard
mysql · Gaps & Islands pattern
-- Logins(user_id, login_date)

WITH deduped AS (
    SELECT DISTINCT user_id, login_date FROM Logins
),
grouped AS (
    -- Subtracting row_number from date groups consecutive dates together
    SELECT
        user_id,
        login_date,
        DATE_SUB(
            login_date,
            INTERVAL ROW_NUMBER() OVER (
                PARTITION BY user_id ORDER BY login_date
            ) DAY
        ) AS grp          -- same grp = consecutive day streak
    FROM deduped
),
streaks AS (
    SELECT user_id, grp, COUNT(*) AS streak_len
    FROM grouped
    GROUP BY user_id, grp
)
SELECT user_id, MAX(streak_len) AS longest_streak
FROM streaks
GROUP BY user_id
ORDER BY longest_streak DESC;

2 Median Employee Salary Hard LC #569

MySQL has no MEDIAN(). Use the double row-number trick to identify the middle row(s).

mysql · LC 569
-- Employee(id, company, salary)

WITH ranked AS (
    SELECT
        id, company, salary,
        ROW_NUMBER() OVER (PARTITION BY company ORDER BY salary)     AS rn,
        COUNT(*)     OVER (PARTITION BY company)                     AS cnt
    FROM Employee
)
SELECT id, company, salary
FROM ranked
-- For odd cnt: single middle row; for even cnt: two middle rows
WHERE rn BETWEEN cnt / 2.0 AND cnt / 2.0 + 1;
Recursive CTEWITH RECURSIVE Recursive CTEs — Hierarchies & Sequences
⚡ Structure: Anchor (base case) + UNION ALL + Recursive (self-references CTE). Always add a depth/level counter to prevent infinite loops.
Syntax Template
sql · skeleton
WITH RECURSIVE cte AS (
    -- Anchor: base case, no self-reference
    SELECT id, parent_id, name, 1 AS depth
    FROM tbl
    WHERE parent_id IS NULL         -- root nodes

    UNION ALL

    -- Recursive: join table back to CTE
    SELECT t.id, t.parent_id, t.name, c.depth + 1
    FROM tbl t
    JOIN cte c ON t.parent_id = c.id
    WHERE c.depth < 20              -- ← always add depth limit!
)
SELECT * FROM cte ORDER BY depth;
1 Generate a Date Series (Fill Missing Dates) Medium

Recursive CTEs can generate number or date sequences — perfect for filling gaps in sparse date data.

mysql · date sequence generator
-- Generate every date in Jan 2024, fill zeros for days with no orders

WITH RECURSIVE cal AS (
    SELECT DATE('2024-01-01') AS dt
    UNION ALL
    SELECT DATE_ADD(dt, INTERVAL 1 DAY)
    FROM cal
    WHERE dt < '2024-01-31'
)
SELECT
    cal.dt,
    COALESCE(SUM(o.amount), 0) AS daily_revenue
FROM cal
LEFT JOIN Orders o ON DATE(o.created_at) = cal.dt
GROUP BY cal.dt
ORDER BY cal.dt;
2 Full Manager Hierarchy / Org Chart Hard
mysql · recursive hierarchy
-- Employee(id, name, manager_id, salary)
-- Find all employees under manager id = 5 (any depth)

WITH RECURSIVE org AS (
    -- Anchor: start from the target manager
    SELECT id, name, manager_id, salary, 0 AS depth
    FROM Employee
    WHERE id = 5

    UNION ALL

    -- Recursive: find direct reports of current level
    SELECT e.id, e.name, e.manager_id, e.salary, o.depth + 1
    FROM Employee e
    JOIN org o ON e.manager_id = o.id
)
SELECT
    CONCAT(REPEAT('  ', depth), name)  AS hierarchy,
    depth,
    salary
FROM org
ORDER BY depth, name;
