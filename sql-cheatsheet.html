SELECT & Filtering (Query Dasar & Penyaringan)

Pondasi setiap query SQL — memilih kolom, menyaring data, dan pengkondisian.

WHEREIS NULLCASE WHEN Penanganan NULL & Penyaringan Kondisional

Template Sintaks
sql
SELECT col1, col2
FROM tbl
WHERE condition1
  AND (condition2 OR condition3)
  AND col IS NOT NULL;   -- selalu gunakan IS NULL, jangan = NULL

col IS NULL              -- mencakup nilai NULL
col IS NOT NULL          -- mengabaikan nilai NULL
COALESCE(col, default)   -- ganti NULL dengan nilai cadangan (fallback)
IFNULL(col, default)     -- alias MySQL untuk COALESCE 2 argumen

SELECT

1 Gaji Tertinggi Kedua (Second Highest Salary) Medium LC #176

Kembalikan NULL jika tidak ada gaji tertinggi kedua — bukan hasil tabel kosong!

-- Tabel Employee(id, salary)

-- Metode 1: Trik Subquery (SELECT luar mengembalikan NULL jika subquery dalam kosong)

    SELECT DISTINCT salary
    FROM Employee
    ORDER BY salary DESC
    LIMIT 1 OFFSET 1
) AS SecondHighestSalary;

Aggregations & GROUP BY (Fungsi Agregasi)

Merangkum dan mengelompokkan data — pola query SQL LeetCode paling sering diuji.

COUNTSUMAVGGROUP BYHAVING Fungsi Agregasi & Pengelompokan Data (GROUP BY)
⚠️ Perbedaan HAVING vs WHERE: WHERE menyaring baris sebelum pengelompokan. HAVING menyaring grup setelah agregasi dilakukan. Kamu tidak bisa menggunakan fungsi agregasi di dalam WHERE.
Template Sintaks
sql
SELECT
    group_col,
    COUNT(*)               AS total_rows,
    COUNT(col)             AS non_null_count,   -- mengabaikan NULL
    COUNT(DISTINCT col)    AS unique_count,
    SUM(amount)            AS total,
    AVG(amount)            AS average,
    MIN(amount)            AS minimum,
    MAX(amount)            AS maximum,
    -- Trik COUNT kondisional (TRUE=1, FALSE=0 di MySQL):
    SUM(col = 'value')     AS matching_count,
    SUM(CASE WHEN col = 'value' THEN amount ELSE 0 END) AS cond_sum
FROM tbl
WHERE pre_filter_condition         -- menyaring SEBELUM GROUP BY
GROUP BY group_col
HAVING aggregate_condition         -- menyaring SETELAH GROUP BY
ORDER BY total_rows DESC;

2 Ringkasan Transaksi Bulanan (Monthly Transactions) Medium LC #1193

Agregasi banyak kolom dengan penjumlahan kondisional — pola yang sangat sering muncul pada soal database.

mysql · LC 1193
-- Tabel Transactions(id, country, state, amount, trans_date)
-- state ∈ {'approved', 'declined'}

SELECT
    DATE_FORMAT(trans_date, '%Y-%m')          AS month,
    country,
    COUNT(*)                                   AS trans_count,
    SUM(state = 'approved')                    AS approved_count,    -- trik boolean
    SUM(amount)                                AS trans_total_amount,
    SUM(CASE WHEN state = 'approved'
             THEN amount ELSE 0 END)           AS approved_total_amount
FROM Transactions
GROUP BY month, country;

2 Departemen dengan Rata-rata Gaji di Atas Rata-rata Perusahaan Medium

Menggabungkan subquery skalar di dalam klausa HAVING — penulisan query yang sangat rapi.

mysql
-- Tabel Employee(id, name, salary, dept_id)

SELECT dept_id, ROUND(AVG(salary), 2) AS avg_salary
FROM Employee
GROUP BY dept_id
HAVING AVG(salary) > (

ORDER BY avg_salary DESC;

JOINs (Penggabungan Tabel)

Topik SQL yang paling sering diuji dalam interview. Kuasai setiap jenis JOIN secara mendalam.

Panduan Ringkas Jenis JOIN
INNER JOIN
Hanya baris yang cocok di kedua tabel
LEFT JOIN
Semua baris kiri + yang cocok di kanan (NULL jika tidak ada)
RIGHT JOIN
Semua baris kanan + yang cocok di kiri (NULL jika tidak ada)
CROSS JOIN
Perkalian Kartesius — setiap baris × setiap baris
SELF JOIN
Tabel digabungkan ke dirinya sendiri dengan dua alias
LEFT JOININNER JOINAnti-Join JOIN Fundamentals & Pola Anti-Join
⚡ Pola Anti-Join: LEFT JOIN … WHERE right.id IS NULL mencari baris yang ada di tabel kiri tetapi tidak memiliki pasangan cocok di tabel kanan. Lebih cepat & aman dari NOT IN ketika ada nilai NULL.
Template Sintaks
sql
-- INNER JOIN: hanya baris yang cocok di kedua tabel
SELECT a.col, b.col
FROM A a INNER JOIN B b ON a.id = b.a_id;

-- LEFT JOIN: semua baris dari A, bernilai NULL di B jika tidak cocok
SELECT a.col, b.col
FROM A a LEFT JOIN B b ON a.id = b.a_id;

-- Anti-Join: baris di A yang TIDAK memiliki pasangan di B
SELECT a.col
FROM A a
LEFT JOIN B b ON a.id = b.a_id
WHERE b.a_id IS NULL;         -- ← filter anti-join

2 Suhu Kenaikan (Rising Temperature) Easy LC #197

Cari tanggal di mana suhu lebih tinggi dari hari sebelumnya. Gunakan DATE_ADD untuk mencocokkan baris kemarin.

mysql · LC 197
-- Weather(id, recordDate, temperature)

SELECT w1.id
FROM Weather w1
JOIN Weather w2
  ON w1.recordDate = DATE_ADD(w2.recordDate, INTERVAL 1 DAY)
WHERE w1.temperature > w2.temperature;

-- DATE_ADD(date, INTERVAL n DAY) = tambah n hari ke tanggal
-- DATE_SUB(date, INTERVAL n DAY) = kurangi n hari dari tanggal

Subqueries & CTEs (Common Table Expressions)

Membangun query yang modular dan mudah dibaca. CTE adalah alat terbaikmu untuk soal bertingkat yang kompleks.

INEXISTSNOT EXISTSCorrelated Subqueries: IN vs EXISTS
⚠️ Jebakan NOT IN + NULL: Jika hasil subquery mengandung nilai NULL, operasi NOT IN akan mengembalikan nol baris! Gunakan NOT EXISTS atau pola Anti-Join untuk keamanan.
Template Sintaks
sql
-- Subquery skalar (mengembalikan 1 nilai tunggal)
WHERE salary = (SELECT MAX(salary) FROM Employee)

-- Subquery IN (mengembalikan daftar nilai)
WHERE dept_id IN (SELECT id FROM Dept WHERE budget > 50000)

-- EXISTS (TRUE jika subquery memiliki minimal 1 baris — sering lebih cepat)
WHERE EXISTS (
    SELECT 1 FROM Orders WHERE Orders.customer_id = Customer.id
)

-- NOT EXISTS (lebih aman daripada NOT IN jika ada NULL)
WHERE NOT EXISTS (
    SELECT 1 FROM Orders WHERE Orders.customer_id = Customer.id
)

-- Correlated subquery (merujuk ke query luar baris demi baris)
WHERE salary > (
    SELECT AVG(salary)
    FROM Employee e2
    WHERE e2.dept_id = e1.dept_id   -- ← merujuk alias luar e1
)

-- Perbandingan Tuple / Baris (Ekstensi MySQL)
WHERE (dept_id, salary) IN (SELECT dept_id, MAX(salary) FROM Employee GROUP BY dept_id)
1 Gaji Tertinggi di Setiap Departemen Medium LC #184

Trik tuple (dept_id, salary) IN (...) menyaring karyawan dengan gaji tertinggi di tiap departemen dengan sangat elegan.

mysql · LC 184
-- Employee(id, name, salary, departmentId)
-- Department(id, name)

SELECT d.name AS Department, e.name AS Employee, e.salary AS Salary
FROM Employee e
JOIN Department d ON e.departmentId = d.id
WHERE (e.departmentId, e.salary) IN (
    SELECT departmentId, MAX(salary)
    FROM Employee
    GROUP BY departmentId
);
2 Trips & Users — CTE Bertingkat Hard LC #262

CTE memungkinkan kita memberi nama pada hasil antara untuk memecah masalah sulit menjadi langkah-langkah yang jelas.

mysql · LC 262 — Pendekatan CTE
-- Trips(id, client_id, driver_id, status, request_at)
-- Users(users_id, banned, role)

WITH non_banned AS (
    -- Langkah 1: Kumpulkan ID user yang tidak di-ban
    SELECT users_id FROM Users WHERE banned = 'No'
),
valid_trips AS (
    -- Langkah 2: Filter trip yang hanya melibatkan user non-banned dalam rentang tanggal
    SELECT *
    FROM Trips
    WHERE request_at BETWEEN '2013-10-01' AND '2013-10-03'
      AND client_id IN (SELECT users_id FROM non_banned)
      AND driver_id IN (SELECT users_id FROM non_banned)
)
-- Langkah 3: Hitung tingkat pembatalan (cancellation rate) harian
SELECT
    request_at                                   AS Day,
    ROUND(SUM(status != 'completed') / COUNT(*), 2) AS 'Cancellation Rate'
FROM valid_trips
GROUP BY request_at;

Window Functions (Fungsi Jendela / Analisis)

Menghitung analisis antar-baris tanpa menggabungkan/menciutkan baris. Sangat krusial untuk soal peringkat & analitik tingkat Medium/Hard.

Skeleton Sintaks Window Function
sql · skeleton
nama_fungsi() OVER (
    PARTITION BY col           -- menentukan kelompok jendela (opsional)
    ORDER BY col DESC          -- pengurutan di dalam setiap partisi (opsional)
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW  -- rentang bingkai (opsional)
)

-- ── Fungsi Pemeringkatan (Ranking functions) ────────────────────────
ROW_NUMBER()       -- nomor urut unik: 1,2,3,4 (tanpa nilai kembar)
RANK()             -- nilai sama mendapat ranking sama, peringkat lanjut melompat: 1,2,2,4
DENSE_RANK()       -- nilai sama mendapat ranking sama, TANPA lompatan: 1,2,2,3  ← Pola Tertinggi ke-N

-- ── Fungsi Navigasi (Navigation functions) ──────────────────────────
LAG(col, n, default)    -- mengambil nilai n baris SEBELUM baris saat ini
LEAD(col, n, default)   -- mengambil nilai n baris SETELAH baris saat ini
FIRST_VALUE(col)        -- mengambil nilai pertama dalam partisi jendela
LAST_VALUE(col)         -- mengambil nilai terakhir dalam partisi jendela

-- ── Agregasi sebagai Window (baris tidak diciutkan!) ──────────────────
SUM(col)   OVER (...)   -- total berjalan (running total)
AVG(col)   OVER (...)   -- rata-rata bergerak (rolling average)
COUNT(col) OVER (...)   -- hitungan berjalan (running count)
NTILE(n)   OVER (...)   -- membagi baris ke dalam n kelompok seimbang (1..n)
DENSE_RANKROW_NUMBERPARTITION BY Fungsi Pemeringkatan (Ranking Functions)
⚡ Gunakan DENSE_RANK() untuk mencari "Nilai Tertinggi ke-N" (menangani nilai kembar dengan aman). Gunakan ROW_NUMBER() untuk memilih 1 baris unik / deduplikasi.
1 Pemeringkatan Skor Tanpa Lompatan (Rank Scores) Medium LC #178

Skor yang sama wajib mendapatkan ranking yang sama tanpa ada celah angka lompatan — persis seperti perilaku DENSE_RANK().

mysql · LC 178
-- Tabel Scores(id, score)

SELECT
    score,
    DENSE_RANK() OVER (ORDER BY score DESC) AS 'rank'
FROM Scores
ORDER BY score DESC;

-- Perbandingan:
-- score:       100, 90, 90, 80
-- RANK():        1,  2,  2,  4   (ada lompatan angka setelah nilai sama)
-- DENSE_RANK():  1,  2,  2,  3   (tanpa lompatan — yang diminta LC)
-- ROW_NUMBER():  1,  2,  3,  4   (selalu unik)
2 Top 3 Gaji di Setiap Departemen Hard LC #185

Perintah PARTITION BY departmentId mereset urutan peringkat untuk setiap departemen secara terpisah.

mysql · LC 185
-- Tabel Employee(id, name, salary, departmentId)
-- Tabel Department(id, name)

WITH ranked AS (
    SELECT
        name, salary, departmentId,
        DENSE_RANK() OVER (
            PARTITION BY departmentId
            ORDER BY salary DESC
        ) AS rnk
    FROM Employee
)
SELECT d.name AS Department, r.name AS Employee, r.salary AS Salary
FROM ranked r
JOIN Department d ON r.departmentId = d.id
WHERE r.rnk <= 3;
LAGLEADSUM OVER LAG / LEAD & Total Kumulatif Berjalan
1 Perubahan Page View dari Hari ke Hari Medium

Fungsi LAG(col, 1, 0) mengambil nilai dari baris sebelumnya. Argumen ketiga adalah nilai default jika tidak ada baris sebelumnya.

mysql · LAG pattern
-- Tabel Activity(user_id, activity_date, page_views)

SELECT
    user_id,
    activity_date,
    page_views,
    LAG(page_views, 1, 0) OVER (
        PARTITION BY user_id
        ORDER BY activity_date
    )                        AS prev_day_views,
    page_views - LAG(page_views, 1, 0) OVER (
        PARTITION BY user_id
        ORDER BY activity_date
    )                        AS daily_change
FROM Activity;

-- LAG(col, offset, default_jika_tidak_ada_baris_sebelumnya)
-- LEAD(col, offset, default_jika_tidak_ada_baris_setelahnya)  ← sama tapi melihat ke depan
2 Pendapatan Kumulatif & Rata-rata Bergerak 7 Hari Medium
mysql · SUM OVER pattern
-- Tabel Sales(sale_id, sale_date, amount)

SELECT
    sale_id,
    sale_date,
    amount,
    -- Total berjalan dari awal hingga baris saat ini
    SUM(amount) OVER (
        ORDER BY sale_date
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    )                        AS running_total,
    -- Rata-rata bergerak 7 hari (baris saat ini + 6 baris sebelumnya)
    ROUND(AVG(amount) OVER (
        ORDER BY sale_date
        ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
    ), 2)                    AS rolling_7day_avg
FROM Sales
ORDER BY sale_date;

2 Ringkasan Durasi Sesi Bulanan Medium
mysql · aritmatika tanggal
-- Tabel Sessions(session_id, user_id, start_time, end_time)

SELECT
    user_id,
    DATE_FORMAT(start_time, '%Y-%m')                          AS month,
    COUNT(*)                                                   AS sessions,
    SUM(TIMESTAMPDIFF(MINUTE, start_time, end_time))          AS total_minutes,
    ROUND(AVG(TIMESTAMPDIFF(MINUTE, start_time, end_time)), 1) AS avg_minutes
FROM Sessions
WHERE start_time >= '2024-01-01'
GROUP BY user_id, DATE_FORMAT(start_time, '%Y-%m')
ORDER BY month, user_id;

SET Operations (Operasi Himpunan)

Menggabungkan, mencari irisan, atau mengurangkan hasil dari beberapa query SQL.

UNIONUNION ALLINTERSECTEXCEPT UNION, INTERSECT & EXCEPT
⚡ Catatan Versi MySQL: Operator INTERSECT dan EXCEPT baru tersedia di MySQL 8.0+. Untuk versi lebih lama, simulasikan dengan INNER JOIN atau LEFT JOIN … IS NULL.
Template Sintaks
sql
-- UNION: gabungkan hasil, hapus duplikat
SELECT col FROM A
UNION
SELECT col FROM B;

-- UNION ALL: gabungkan hasil, PERTAHANKAN duplikat (lebih cepat, gunakan jika yakin tak ada duplikat)
SELECT col FROM A
UNION ALL
SELECT col FROM B;

-- INTERSECT: ambil baris yang ada di KEDUA hasil query (MySQL 8.0+)
SELECT col FROM A
INTERSECT
SELECT col FROM B;

-- EXCEPT: ambil baris di query pertama yang TIDAK ada di query kedua (MySQL 8.0+)
SELECT col FROM A
EXCEPT
SELECT col FROM B;

-- Syarat: jumlah kolom harus sama, tipe data tiap posisi harus kompatibel
1 Gabungkan Data Karyawan & Kontraktor Easy
mysql · UNION pattern
-- Tabel Employees(id, name, email)  Contractors(id, name, email)

SELECT email, 'employee'   AS type FROM Employees
UNION                                 -- menghapus duplikasi email
SELECT email, 'contractor' AS type FROM Contractors
ORDER BY type, email;

-- Gunakan UNION ALL jika email yang sama sah muncul di kedua tabel
-- dan kamu ingin menampilkan semua baris secara utuh
2 Mencari Non-Teman (Simulasi EXCEPT) Medium

Mencari "semua orang kecuali kelompok tertentu" — menampilkan sintaks MySQL 8.0+ EXCEPT dan Anti-Join bawaan.

mysql · EXCEPT & fallback
-- Tabel Users(id, name)  Friends(user1_id, user2_id)
-- Cari pengguna yang BUKAN teman dari user_id = 1

-- Sintaks MySQL 8.0+:
SELECT id, name FROM Users WHERE id != 1
EXCEPT
SELECT u.id, u.name
FROM Users u
JOIN Friends f
  ON (f.user1_id = 1 AND f.user2_id = u.id)
  OR (f.user2_id = 1 AND f.user1_id = u.id);

-- Anti-join kompatibel (berfungsi di semua versi MySQL):
SELECT u.id, u.name
FROM Users u
LEFT JOIN Friends f
  ON (f.user1_id = 1 AND f.user2_id = u.id)
  OR (f.user2_id = 1 AND f.user1_id = u.id)
WHERE u.id != 1
  AND f.user1_id IS NULL;   -- tidak ada hubungan pertemanan

Gaps & IslandsConsecutiveStreak Baris Berurutan & Deteksi Rekor Beruntun (Streak)
⚡ Trik Klasik Gaps & Islands: Kurangi ROW_NUMBER() dari urutan tanggal — baris yang berada dalam kelompok berurutan yang sama akan menghasilkan nilai selisih yang sama (kunci "pulau/island").
1 Angka Berurutan Minimal 3 Baris (Consecutive Numbers) Medium LC #180
mysql · LC 180
-- Tabel Logs(id, num)   id bersifat sekuensial

-- Metode 1: Triple Self-Join (sederhana, khusus tepat 3 baris)
SELECT DISTINCT l1.num AS ConsecutiveNums
FROM Logs l1
JOIN Logs l2 ON l2.id = l1.id + 1 AND l2.num = l1.num
JOIN Logs l3 ON l3.id = l1.id + 2 AND l3.num = l1.num;

-- Metode 2: LAG/LEAD (dapat diskalakan ke N baris berurutan)
WITH flagged AS (
    SELECT num,
           LAG(num, 1) OVER (ORDER BY id) AS p1,
           LAG(num, 2) OVER (ORDER BY id) AS p2
    FROM Logs
)
SELECT DISTINCT num AS ConsecutiveNums
FROM flagged
WHERE num = p1 AND num = p2;
2 Rekor Login Beruntun Terpanjang (Longest Login Streak) Hard
mysql · Gaps & Islands pattern
-- Tabel Logins(user_id, login_date)

WITH deduped AS (
    SELECT DISTINCT user_id, login_date FROM Logins
),
grouped AS (
    -- Pengurangan row_number dari tanggal mengelompokkan tanggal berurutan menjadi 1 grup
    SELECT
        user_id,
        login_date,
        DATE_SUB(
            login_date,
            INTERVAL ROW_NUMBER() OVER (
                PARTITION BY user_id ORDER BY login_date
            ) DAY
        ) AS grp          -- grp sama = rekor hari berurut yang sama
    FROM deduped
),
streaks AS (
    SELECT user_id, grp, COUNT(*) AS streak_len
    FROM grouped
    GROUP BY user_id, grp
)
SELECT user_id, MAX(streak_len) AS longest_streak
FROM streaks
GROUP BY user_id
ORDER BY longest_streak DESC;
PIVOTConditional Agg Mengubah Baris Menjadi Kolom (Pivot)
⚡ MySQL tidak memiliki kata kunci PIVOT bawaan. Simulasikan dengan SUM(CASE WHEN col = 'value' THEN amount END) — satu kolom untuk setiap nilai pivot.
1 Format Ulang Tabel Departemen (Pivot Bulanan) Medium LC #1179
mysql · LC 1179
-- Tabel Department(id, revenue, month)
-- month ∈ {'Jan','Feb',...,'Dec'}
-- Pivot: 1 baris per id, 1 kolom per bulan

SELECT
    id,
    SUM(CASE WHEN month = 'Jan' THEN revenue END) AS Jan_Revenue,
    SUM(CASE WHEN month = 'Feb' THEN revenue END) AS Feb_Revenue,
    SUM(CASE WHEN month = 'Mar' THEN revenue END) AS Mar_Revenue,
    SUM(CASE WHEN month = 'Apr' THEN revenue END) AS Apr_Revenue,
    SUM(CASE WHEN month = 'May' THEN revenue END) AS May_Revenue,
    SUM(CASE WHEN month = 'Jun' THEN revenue END) AS Jun_Revenue,
    SUM(CASE WHEN month = 'Jul' THEN revenue END) AS Jul_Revenue,
    SUM(CASE WHEN month = 'Aug' THEN revenue END) AS Aug_Revenue,
    SUM(CASE WHEN month = 'Sep' THEN revenue END) AS Sep_Revenue,
    SUM(CASE WHEN month = 'Oct' THEN revenue END) AS Oct_Revenue,
    SUM(CASE WHEN month = 'Nov' THEN revenue END) AS Nov_Revenue,

2 Mencari Median Gaji Karyawan Hard LC #569

MySQL tidak memiliki fungsi MEDIAN(). Gunakan trik ganda ROW_NUMBER() dan COUNT(*) untuk mengidentifikasi baris tengah.

mysql · LC 569
-- Tabel Employee(id, company, salary)

WITH ranked AS (
    SELECT
        id, company, salary,
        ROW_NUMBER() OVER (PARTITION BY company ORDER BY salary)     AS rn,
        COUNT(*)     OVER (PARTITION BY company)                     AS cnt
    FROM Employee
)
SELECT id, company, salary
FROM ranked
-- Jika total genap: 2 baris tengah; Jika ganjil: 1 baris tengah tepat
WHERE rn BETWEEN cnt / 2.0 AND cnt / 2.0 + 1;
Recursive CTEWITH RECURSIVE Recursive CTE — Hirarki & Deret Sekuensial
⚡ Struktur: Anchor (kasus dasar) + UNION ALL + Rekursif (merujuk ke CTE itu sendiri). Wajib menambahkan pembatas kedalaman (depth limit) untuk mencegah perulangan tak terbatas.
Template Sintaks
sql · skeleton
WITH RECURSIVE cte AS (
    -- Anchor: kasus dasar (base case), tanpa referensi ke diri sendiri
    SELECT id, parent_id, name, 1 AS depth
    FROM tbl
    WHERE parent_id IS NULL         -- node akar (root)

    UNION ALL

    -- Rekursif: gabungkan tabel kembali ke CTE
    SELECT t.id, t.parent_id, t.name, c.depth + 1
    FROM tbl t
    JOIN cte c ON t.parent_id = c.id
    WHERE c.depth < 20              -- ← selalu berikan batas kedalaman!
)
SELECT * FROM cte ORDER BY depth;
1 Membuat Deret Tanggal (Melengkapi Tanggal Kosong) Medium

Recursive CTE dapat menggenerasi deret angka atau tanggal — sangat sempurna untuk mengisi celah tanggal yang tidak memiliki transaksi.

mysql · date sequence generator
-- Generasi setiap tanggal pada Januari 2024, beri nilai 0 jika tidak ada pesanan

WITH RECURSIVE cal AS (
    SELECT DATE('2024-01-01') AS dt
    UNION ALL
    SELECT DATE_ADD(dt, INTERVAL 1 DAY)
    FROM cal
    WHERE dt < '2024-01-31'
)
SELECT
    cal.dt,
    COALESCE(SUM(o.amount), 0) AS daily_revenue
FROM cal
LEFT JOIN Orders o ON DATE(o.created_at) = cal.dt
GROUP BY cal.dt
ORDER BY cal.dt;
2 Struktur Hirarki Manajer (Org Chart) Hard
mysql · recursive hierarchy
-- Tabel Employee(id, name, manager_id, salary)
-- Cari semua bawahan di bawah manajer id = 5 (sampai kedalaman berapapun)

WITH RECURSIVE org AS (
    -- Anchor: mulai dari manajer yang dituju
    SELECT id, name, manager_id, salary, 0 AS depth
    FROM Employee
    WHERE id = 5

    UNION ALL

    -- Rekursif: cari bawahan langsung dari level saat ini
    SELECT e.id, e.name, e.manager_id, e.salary, o.depth + 1
    FROM Employee e
    JOIN org o ON e.manager_id = o.id
)
SELECT
    CONCAT(REPEAT('  ', depth), name)  AS hierarchy,
    depth,
    salary
FROM org
ORDER BY depth, name;
